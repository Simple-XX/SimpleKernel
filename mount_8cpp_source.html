180 // 确保是完整匹配或下一个字符是 /，或者是根目录挂载
181 char next_char = path[mp_len];
182 if (next_char == '\0' || next_char == '/' ||
183 (mp_len == 1 && mp_path[0] == '/')) {
184 // 选择最长的匹配

186 best_match = &mounts_[i];
187 best_match_len = mp_len;
